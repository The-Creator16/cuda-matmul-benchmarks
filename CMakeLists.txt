cmake_minimum_required(VERSION 3.24)
project(cuda_matmul_benchmarks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_CUDA "Build the GPU benchmark (requires CUDA Toolkit and cuBLAS)" ON)
enable_testing()
add_executable(host_tests tests/host_tests.cpp)
target_include_directories(host_tests PRIVATE include)
add_test(NAME host_tests COMMAND host_tests)

if(BUILD_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES native)
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit 11.0 REQUIRED)
    add_executable(matmul_bench src/main.cu)
    set_target_properties(matmul_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_include_directories(matmul_bench PRIVATE include)
    target_link_libraries(matmul_bench PRIVATE CUDA::cudart CUDA::cublas)
    target_compile_options(matmul_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
    # Deliberately no --use_fast_math: compare FP32 implementations.
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME gpu_correctness COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate.py --binary $<TARGET_FILE:matmul_bench>)
        set_tests_properties(gpu_correctness PROPERTIES LABELS "gpu" TIMEOUT 600)
    endif()
endif()
